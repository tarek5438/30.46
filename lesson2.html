box = '⬜️'
x = '❌'
o = '⭕️'

row1 = [box, box, box]
row2 = [box, box, box]
row3 = [box, box, box]

print(f"{' '.join(row1)}\n{' '.join(row2)}\n{' '.join(row3)}")


game = [row1, row2, row3]
while True:
    # player x.
    px = input("Enter a position for x: ")
    row_number = int(px[0])-1
    item_number = int(px[1])-1

    while game[row_number][item_number] != box:
        px = input("Enter a new empty position for x: ")
        row_number = int(px[0])-1
        item_number = int(px[1])-1

    game[row_number][item_number] = x

    print(f"{' '.join(row1)}\n{' '.join(row2)}\n{' '.join(row3)}")
    # winning of player x.
    if row1[0] == x and row1[1] == x and row1[2] == x:
        print('player x is the winner')
        break

    # player o.
    po = input("Enter a position for o: ")
    row_number = int(po[0])-1
    item_number = int(po[1])-1

    # loop
    while game[row_number][item_number] != box:
        po = input("Enter a new empty position for o: ")
        row_number = int(po[0])-1
        item_number = int(po[1])-1

    game[row_number][item_number] = o

    print(f"{' '.join(row1)}\n{' '.join(row2)}\n{' '.join(row3)}")
